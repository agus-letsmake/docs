<Warning>
  For CI reporting use
  [`@ovenbits/make-testlab-playwright`](/docs/cli-sdk/playwright) or
  [`@ovenbits/make-testlab-client`](/docs/cli-sdk/client). Machine calls need
  **both** `X-API-Key` and `X-Project-Id`. Automated results match cases by
  **`automationKey`**, not Playwright titles.
</Warning>

See [Quick start](/docs/guides/quickstart) and
[GitHub Actions](/docs/ci-automations/github-actions) for the current path.

The snippets below are raw HTTP helpers (bulk case create, export/import).
They are not a substitute for `startExecution` → reporter → `completeExecution`.

## Creating Test Cases in Bulk

Create multiple test cases at once:

```javascript
const testCases = [
  {
    title: "User can login with valid credentials",
    steps: [
      {
        id: "step1",
        description: "Navigate to login",
        expectedResult: "Login form shown",
      },
      {
        id: "step2",
        description: "Enter credentials",
        expectedResult: "Credentials accepted",
      },
      {
        id: "step3",
        description: "Click login",
        expectedResult: "Redirected to dashboard",
      },
    ],
    labels: ["authentication", "critical"],
    automationStatus: "automated",
  },
  {
    title: "User cannot login with invalid credentials",
    steps: [
      {
        id: "step1",
        description: "Navigate to login",
        expectedResult: "Login form shown",
      },
      {
        id: "step2",
        description: "Enter invalid credentials",
        expectedResult: "Error shown",
      },
    ],
    labels: ["authentication", "critical"],
    automationStatus: "automated",
  },
];

for (const testCase of testCases) {
  const response = await fetch("https://testlab.letsmake.com/api/test-cases", {
    method: "POST",
    headers: {
      "X-API-Key": process.env.TESTLAB_API_KEY,
      "X-Project-Id": process.env.TESTLAB_PROJECT_ID,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(testCase),
  });

  const created = await response.json();
  console.log(`Created: ${created.id} - ${created.title}`);
}
```

## Searching Test Cases by Label

```javascript
// Get all test cases
const response = await fetch("https://testlab.letsmake.com/api/test-cases", {
  headers: { "X-API-Key": process.env.TESTLAB_API_KEY },
});

const allCases = await response.json();

// Filter by label
const criticalTests = allCases.filter((tc) => tc.labels.includes("critical"));

console.log(`Found ${criticalTests.length} critical test cases`);
```

## Creating a Test Suite from Filtered Cases

```javascript
// Get all automated test cases
const response = await fetch("https://testlab.letsmake.com/api/test-cases", {
  headers: { "X-API-Key": process.env.TESTLAB_API_KEY },
});

const allCases = await response.json();
const automatedCases = allCases.filter(
  (tc) => tc.automationStatus === "automated",
);

// Create suite with automated tests
const suiteResponse = await fetch(
  "https://testlab.letsmake.com/api/test-suites",
  {
    method: "POST",
    headers: {
      "X-API-Key": process.env.TESTLAB_API_KEY,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      name: "Automated Test Suite",
      description: "All automated tests for CI/CD",
      testCaseIds: automatedCases.map((tc) => tc.id),
    }),
  },
);

const suite = await suiteResponse.json();
console.log(`Created suite with ${suite.testCaseIds.length} test cases`);
```

## Running a Complete Test Execution

```javascript
async function runTestExecution(suiteId, suiteName, testCaseIds) {
  // 1. Create execution
  const execResponse = await fetch(
    "https://testlab.letsmake.com/api/executions",
    {
      method: "POST",
      headers: {
        "X-API-Key": process.env.TESTLAB_API_KEY,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        suiteId,
        suiteName,
        testCaseIds,
        name: `Manual Test Run - ${new Date().toLocaleDateString()}`,
        environment: "manual",
      }),
    },
  );

  const execution = await execResponse.json();
  console.log(`Execution created: ${execution.id}`);

  // 2. Update results for each test case
  const results = [];
  for (const testCaseId of testCaseIds) {
    // Simulate manual testing
    const passed = Math.random() > 0.2; // 80% pass rate

    results.push({
      testCaseId,
      status: passed ? "pass" : "fail",
      notes: passed ? "All steps completed successfully" : "Failed on step 2",
    });
  }

  await fetch(
    `https://testlab.letsmake.com/api/executions/${execution.id}/results`,
    {
      method: "POST",
      headers: {
        "X-API-Key": process.env.TESTLAB_API_KEY,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ results }),
    },
  );

  // 3. Mark execution as complete
  await fetch(`https://testlab.letsmake.com/api/executions/${execution.id}`, {
    method: "PUT",
    headers: {
      "X-API-Key": process.env.TESTLAB_API_KEY,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({ status: "completed" }),
  });

  console.log("Execution completed!");
  return execution.id;
}
```

## Generating Test Metrics Report

```javascript
async function generateMetrics() {
  // Get all executions
  const execResponse = await fetch(
    "https://testlab.letsmake.com/api/executions",
    {
      headers: { "X-API-Key": process.env.TESTLAB_API_KEY },
    },
  );

  const executions = await execResponse.json();

  // Calculate metrics
  const metrics = {
    totalExecutions: executions.length,
    completedExecutions: executions.filter((e) => e.status === "completed")
      .length,
    totalTests: 0,
    passedTests: 0,
    failedTests: 0,
    automatedResults: 0,
    manualResults: 0,
  };

  executions.forEach((exec) => {
    exec.testResults.forEach((result) => {
      metrics.totalTests++;
      if (result.status === "pass") metrics.passedTests++;
      if (result.status === "fail") metrics.failedTests++;
      if (result.source === "automated") metrics.automatedResults++;
      if (result.source === "manual") metrics.manualResults++;
    });
  });

  metrics.passRate = ((metrics.passedTests / metrics.totalTests) * 100).toFixed(
    2,
  );
  metrics.automationRate = (
    (metrics.automatedResults / metrics.totalTests) *
    100
  ).toFixed(2);

  console.log("Test Metrics:");
  console.log(`  Total Executions: ${metrics.totalExecutions}`);
  console.log(`  Pass Rate: ${metrics.passRate}%`);
  console.log(`  Automation Rate: ${metrics.automationRate}%`);

  return metrics;
}
```

## Exporting Test Data for Backup

```javascript
async function backupTestData() {
  const response = await fetch("https://testlab.letsmake.com/api/export", {
    headers: { "X-API-Key": process.env.TESTLAB_API_KEY },
  });

  const data = await response.json();

  // Save to file
  const fs = require("fs");
  const timestamp = new Date().toISOString().split("T")[0];
  const filename = `testlab-backup-${timestamp}.json`;

  fs.writeFileSync(filename, JSON.stringify(data, null, 2));
  console.log(`Backup saved to ${filename}`);
  console.log(`  Test Cases: ${data.testCases.length}`);
  console.log(`  Test Suites: ${data.testSuites.length}`);
}
```

## Importing Test Data

```javascript
async function importTestData(filename) {
  const fs = require("fs");
  const data = JSON.parse(fs.readFileSync(filename, "utf8"));

  const response = await fetch("https://testlab.letsmake.com/api/import", {
    method: "POST",
    headers: {
      "X-API-Key": process.env.TESTLAB_API_KEY,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(data),
  });

  const result = await response.json();
  console.log("Import completed:");
  console.log(`  Test Cases: ${result.imported.testCases}`);
  console.log(`  Test Suites: ${result.imported.testSuites}`);
}
```

## Webhook for Slack Notifications

```javascript
async function sendSlackNotification(executionId) {
  // Get execution results
  const response = await fetch(
    `https://testlab.letsmake.com/api/automated-results?executionId=${executionId}`,
    {
      headers: { "X-API-Key": process.env.TESTLAB_API_KEY },
    },
  );

  const data = await response.json();
  const { summary } = data;

  // Determine color based on results
  const color = summary.fail > 0 ? "danger" : "good";
  const emoji = summary.fail > 0 ? "❌" : "✅";

  // Send to Slack
  await fetch(process.env.SLACK_WEBHOOK_URL, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      text: `${emoji} Test Results: ${data.name}`,
      attachments: [
        {
          color,
          fields: [
            {
              title: "Environment",
              value: data.environment,
              short: true,
            },
            {
              title: "Version",
              value: data.version,
              short: true,
            },
            {
              title: "Passed",
              value: `${summary.pass}`,
              short: true,
            },
            {
              title: "Failed",
              value: `${summary.fail}`,
              short: true,
            },
            {
              title: "View Results",
              value: `https://testlab.letsmake.com/executions/${executionId}`,
              short: false,
            },
          ],
        },
      ],
    }),
  });
}
```

## Python Example: Automated Test Reporting

Prefer `@ovenbits/make-testlab-client` from Node CI. If you must call HTTP from
Python, start an execution first and push `automationKey` (not `testCaseId`):

```python
import os
import requests

BASE = os.environ["TESTLAB_URL"]
HEADERS = {
    "X-API-Key": os.environ["TESTLAB_API_KEY"],
    "X-Project-Id": os.environ["TESTLAB_PROJECT_ID"],
    "Content-Type": "application/json",
}

started = requests.post(
    f"{BASE}/api/automated-results/start",
    json={"suiteId": os.environ["TESTLAB_SUITE_ID"], "name": "CI"},
    headers=HEADERS,
).json()
execution_id = started["executionId"]

requests.post(
    f"{BASE}/api/automated-results",
    json={
        "executionId": execution_id,
        "results": [
            {"automationKey": "DEMO-01", "status": "pass", "duration": 234},
        ],
    },
    headers=HEADERS,
)
```
